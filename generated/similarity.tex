\begin{cleanresult}{similarity transforms eigenvector}\cleanattribution{[Stewart 2001, Theorem 1.10, page 9]}
Let $n$ be a natural number.
Let $A$ and $S$ be $n\times n$ matrices over $\mathbb{C}$.
Let $x$ be a vector of length $n$ over $\mathbb{C}$.
Let $\mu$ be a complex number.
Assume $\det(S) \ne 0$ (\texttt{invertibility}).
Assume $x \ne 0$ (\texttt{nonzero}).
Assume $A \, x = \mu \, x$ (\texttt{eigenpair}).
Let $y = {S}^{-1} \, x$.
Then \[y \ne 0 \text{ and } {S}^{-1} \, A \, S \, y = \mu \, y.\]

\end{cleanresult}
